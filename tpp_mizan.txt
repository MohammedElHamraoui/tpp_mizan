# Define phony targets (targets that don't represent files)
.PHONY: help sync install test lint format run run-api build clean deep-clean context add add-dev rename

# Default target
help: ## Show this help message
	@echo "Usage: make [target]"
	@echo ""
	@echo "Targets:"
	@grep -E '^[a-zA-Z_-]+:.*?## .*$$' $(MAKEFILE_LIST) | awk 'BEGIN {FS = ":.*?## "}; {printf "  \033[36m%-15s\033[0m %s\n", $$1, $$2}'

# --- Dependency Management ---
sync: ## Create venv, resolve lockfile, and install project
	uv sync

install: sync ## Ensure project is installed (runs sync)
	@echo "Project synced and installed successfully."

add: ## Add a new dependency (Usage: make add PKG=requests)
	uv add $(PKG)

add-dev: ## Add a dev dependency (Usage: make add-dev PKG=httpx2)
	uv add --dev $(PKG)

remove: ## Remove a dependency (Usage: make remove PKG=requests)
	uv remove $(PKG)

lock: ## Update the lockfile without installing to the venv
	uv lock

# --- Code Quality ---
format: ## Format code using Ruff
	uv run ruff format .
	uv run ruff check --fix .

lint: ## Lint code using Ruff and Mypy
	uv run ruff check .
	uv run mypy .

# --- Testing ---
test: ## Run tests with pytest
	uv run pytest -v

test-cov: ## Run tests with coverage report
	uv run pytest --cov --cov-report=term-missing

# --- Execution ---
# Define default values that can be overridden from the command line
X ?= 10
Y ?= 2

run: ## Run the main CLI application (Usage: make run X=5 Y=2)
	uv run my-divide $(X) $(Y)

run-api: ## Run the FastAPI application with hot-reloading
	uv run uvicorn tpp_mizan.app:app --reload --host 127.0.0.1 --port 8000

# --- Maintenance ---
clean: ## Clean up caches and virtual environments
	rm -rf .venv .pytest_cache .mypy_cache .ruff_cache .coverage dist build
	find . -type d -name "__pycache__" -exec rm -rf {} +
	uv cache clean

deep-clean: clean ## Nuclear clean: removes logs, context file, and lockfile
	rm -rf logs/
	rm -f tpp_mizan.txt
	rm -f uv.lock
	@echo "Deep clean complete. (Next 'make install' will re-resolve dependencies)"

# --- CI/CD ---
ci-check: format lint test ## Run all CI checks locally

# --- LLM ---
context: ## Export project source to tpp_mizan.txt for LLM feedback
	@echo "Generating LLM context file..."
	@find . -type f \( -name "*.py" -o -name "*.toml" -o -name "*.md" -o -name "*.yaml" -o -name "Makefile" \) \
		-not -path "./.venv/*" -not -path "./logs/*" -not -path "./.git/*" \
		| sort | while read -r f; do \
			echo "========== $$f =========="; \
			cat "$$f"; \
			echo ""; \
		done > tpp_mizan.txt
	@echo "Done! Saved to tpp_mizan.txt ($$(wc -l < tpp_mizan.txt) lines, $$(du -h tpp_mizan.txt | cut -f1))"

rename: ## Rename the project (Usage: make rename NEW_NAME=tpp_mizan)
	@if [ -z "$(NEW_NAME)" ]; then \
		echo "Error: NEW_NAME is required. Usage: make rename NEW_NAME=tpp_mizan"; \
		exit 1; \
	fi
	@echo "Renaming project to '$(NEW_NAME)' (project: tpp-mizan)..."
	# 1. Update pyproject.toml project name (uses hyphens per PEP 621)
	@sed -i 's/name = "tpp-mizan"/name = "tpp-mizan"/' pyproject.toml
	# 2. Rename source and test directories (uses underscores for Python imports)
	@if [ -d "src/tpp_mizan" ]; then mv src/tpp_mizan src/$(NEW_NAME); fi
	@if [ -d "tests/tpp_mizan" ]; then mv tests/tpp_mizan tests/$(NEW_NAME); fi
	# 3. Replace all internal module references (tpp_mizan -> tpp_mizan)
	@find . -type f \( -name "*.py" -o -name "*.toml" -o -name "*.yaml" -o -name "Makefile" -o -name "*.md" \) \
		-not -path "./.venv/*" -not -path "./.git/*" -exec sed -i "s/tpp_mizan/$(NEW_NAME)/g" {} +
	# 4. Replace all internal project name references (tpp-mizan -> tpp-mizan)
	@find . -type f \( -name "*.py" -o -name "*.toml" -o -name "*.yaml" -o -name "Makefile" -o -name "*.md" \) \
		-not -path "./.venv/*" -not -path "./.git/*" -exec sed -i "s/tpp-mizan/tpp-mizan/g" {} +
	@echo "✅ Rename complete! Run 'make deep-clean && make install' to rebuild the environment."

# My Project

A modern, production-ready Python template featuring FastAPI, Typer CLI, and structured JSON logging, managed by `uv`.

## ✨ Features
- ⚡ **Ultra-fast dependency management** with `uv`
- 🚀 **FastAPI** for high-performance, auto-documented REST APIs
- 💻 **Typer** for type-safe, intuitive CLI tools
- 📝 **Structured JSON Logging** with dynamic path resolution and log rotation
- ✅ **Bulletproof CI/CD** via `make ci-check` (Ruff, Mypy, Pytest)

## 🚀 Quick Start

1. **Install dependencies**:
   ```bash
   make install
   ```

2. **Run the CLI**:
   ```bash
   make run X=10 Y=2
   # or directly: uv run my-divide 10 2
   ```

3. **Start the API**:
   ```bash
   make run-api
   # Visit http://127.0.0.1:8000/docs for the interactive Swagger UI
   ```

4. **Run CI Checks**:
   ```bash
   make ci-check
   ```

## 🛠️ Development
- Add a dependency: `make add PKG=requests`
- Add a dev dependency: `make add-dev PKG=httpx2`
- Clean up: `make clean` (or `make deep-clean` for a total reset)

# src/tpp_mizan/app.py
import logging

from fastapi import FastAPI, HTTPException

from tpp_mizan.core.logging import configure_loggers
from tpp_mizan.core.math_ops import div

# Initialize logging when the API module is loaded
logger = logging.getLogger(__name__)
configure_loggers()


app = FastAPI(title="My Project API")


@app.get("/")
def read_root():
    logger.info("Root endpoint accessed")
    return {"message": "Welcome to My Project API"}


@app.get("/divide/{x}/{y}")
def divide(x: float, y: float):
    logger.info("API request: divide %s by %s", x, y)
    try:
        result = div(x, y)
        return {"x": x, "y": y, "result": result}
    except ZeroDivisionError as e:
        logger.error("API error: %s", e)
        # Return a proper HTTP 400 error instead of a 500 server crash
        raise HTTPException(status_code=400, detail=str(e))

# src/tpp_mizan/cli.py
import logging

import typer

from tpp_mizan.core.logging import configure_loggers
from tpp_mizan.core.math_ops import div

# Initialize logging when the CLI starts
configure_loggers()
logger = logging.getLogger(__name__)

app = typer.Typer(help="A simple division CLI tool.")


@app.command()
def divide(
    x: float = typer.Argument(..., help="The numerator"),
    y: float = typer.Argument(..., help="The denominator"),
):
    """Divide X by Y."""
    logger.info("CLI request: divide %s by %s", x, y)  # <-- Add this
    try:
        result = div(x, y)
        logger.info("CLI success: %s / %s = %s", x, y, result)  # <-- Add this
        typer.echo(f"Result: {x} / {y} = {result}")
    except ZeroDivisionError as e:
        logger.error("CLI error: %s", e)  # <-- Add this
        typer.echo("Error: Division by zero.", err=True)
        raise typer.Exit(code=1)


if __name__ == "__main__":
    app()

# src/tpp_mizan/core/logging.py
import logging
import logging.config
import os
from pathlib import Path

from yaml import safe_load


def configure_loggers():
    package_dir = Path(__file__).resolve().parent.parent
    config_path = package_dir / "config" / "my_log.yaml"

    with open(config_path) as f:
        config = safe_load(f)

    # Dynamically set log path to project root
    project_root = package_dir.parent.parent
    log_dir = project_root / "logs"
    log_dir.mkdir(parents=True, exist_ok=True)
    config["handlers"]["json_file"]["filename"] = str(log_dir / "tpp_mizan.json.log")
    config["handlers"]["timed_file"]["filename"] = str(
        log_dir / "tpp_mizan_timed.log"
    )  # <-- Added this

    # Override log level from env
    env_log_level = os.getenv("LOG_LEVEL", "DEBUG").upper()
    config["loggers"]["tpp_mizan"]["level"] = env_log_level

    # Also override console handler level
    config["handlers"]["c_handler"]["level"] = env_log_level

    logging.config.dictConfig(config)

# src/tpp_mizan/core/math_ops.py
import logging

logger = logging.getLogger(__name__)


def div(x: float, y: float) -> float:
    """Divide x by y. Raises ZeroDivisionError if y is 0."""
    if y == 0:
        logger.error("A math error occurred: Division by zero!")
        raise ZeroDivisionError("Cannot divide by zero")
    return x / y
