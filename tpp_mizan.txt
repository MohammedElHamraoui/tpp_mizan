========== ./README.md ==========
# TPP Mizan

A production-ready Python template from Tech Print Pack featuring FastAPI REST API, Typer CLI, and structured JSON logging, managed by `uv`.

## ✨ Features
- ⚡ **Ultra-fast dependency management** with `uv`
- 🚀 **FastAPI** for high-performance, auto-documented REST APIs
- 💻 **Typer** for type-safe, intuitive CLI tools
- 📝 **Structured JSON Logging** with dynamic path resolution and log rotation
- ✅ **Bulletproof CI/CD** via GitHub Actions running `make ci-check` (Ruff, Mypy, Pytest)

## 🚀 Quick Start

1. **Install dependencies**:
   ```bash
   make install
   ```

2. **Run the CLI** (supports `add`, `subtract`, `multiply`, `divide`):
   ```bash
   make run-add X=10 Y=2        # → 12.0
   make run-subtract X=10 Y=2   # → 8.0
   make run-multiply X=10 Y=2   # → 20.0
   make run X=10 Y=2            # → 5.0 (divide is the default)
   ```

3. **Start the API**:
   ```bash
   make run-api
   # Visit http://127.0.0.1:8000/docs for the interactive Swagger UI
   # Endpoints: /add/{x}/{y}, /subtract/{x}/{y}, /multiply/{x}/{y}, /divide/{x}/{y}
   ```

4. **Run CI Checks**:
   ```bash
   make ci-check
   ```

## ⚙️ Environment Variables
You can configure the application by setting these variables in your shell or a `.env` file:

| Variable | Default | Description |
|---|---|---|
| `LOG_LEVEL` | `INFO` | Sets the logging verbosity (`DEBUG`, `INFO`, `WARNING`, `ERROR`). |
| `TPP_MIZAN_LOG_DIR` | `./logs` | Overrides the directory where JSON and timed log files are written. |

## 🛠️ Development
- Add a dependency: `make add PKG=requests`
- Add a dev dependency: `make add-dev PKG=httpx2`
- Clean up: `make clean` (or `make deep-clean` for a total reset)
- Generate LLM context: `make context`

# src/tpp_mizan/app.py
import logging
from collections.abc import AsyncIterator
from contextlib import asynccontextmanager

from fastapi import FastAPI, HTTPException

from tpp_mizan.core.logging import configure_loggers
from tpp_mizan.core.math_ops import add, div, multiply, subtract
from tpp_mizan.schemas import MathResult


@asynccontextmanager
async def lifespan(app: FastAPI) -> AsyncIterator[None]:
    configure_loggers()
    yield


logger = logging.getLogger(__name__)
app = FastAPI(title="TPP Mizan API", lifespan=lifespan)


@app.get("/")
def read_root() -> dict[str, str]:
    logger.info("Root endpoint accessed")
    return {"message": "Welcome to TPP Mizan API"}


@app.get("/add/{x}/{y}", response_model=MathResult)
def add_endpoint(x: float, y: float) -> MathResult:
    logger.info("API request: add %s + %s", x, y)
    return MathResult(operation="add", x=x, y=y, result=add(x, y))


@app.get("/subtract/{x}/{y}", response_model=MathResult)
def subtract_endpoint(x: float, y: float) -> MathResult:
    logger.info("API request: subtract %s - %s", x, y)
    return MathResult(operation="subtract", x=x, y=y, result=subtract(x, y))


@app.get("/multiply/{x}/{y}", response_model=MathResult)
def multiply_endpoint(x: float, y: float) -> MathResult:
    logger.info("API request: multiply %s * %s", x, y)
    return MathResult(operation="multiply", x=x, y=y, result=multiply(x, y))


@app.get("/divide/{x}/{y}", response_model=MathResult)
def divide_endpoint(x: float, y: float) -> MathResult:
    logger.info("API request: divide %s / %s", x, y)
    try:
        return MathResult(operation="divide", x=x, y=y, result=div(x, y))
    except ZeroDivisionError as e:
        raise HTTPException(status_code=400, detail=str(e)) from None

# src/tpp_mizan/cli.py
import logging

import typer

from tpp_mizan import __version__
from tpp_mizan.core.logging import configure_loggers
from tpp_mizan.core.math_ops import add, div, multiply, subtract

# Logger is instantiated here, but handlers are attached later by configure_loggers()
logger = logging.getLogger(__name__)

app = typer.Typer(
    help="TPP Mizan: A production-ready math CLI tool for Tech Print Pack.",
    no_args_is_help=True,
    add_completion=False,
)


def version_callback(value: bool) -> None:
    if value:
        typer.echo(f"tpp-mizan {__version__}")
        raise typer.Exit()


@app.callback()
def main(
    version: bool = typer.Option(
        False,  # <-- Changed from None to False to satisfy mypy strict mode
        "--version",
        # "-v" removed: reserved for future --verbose logging flag
        callback=version_callback,
        is_eager=True,
        help="Show the version and exit.",
    ),
) -> None:
    """TPP Mizan CLI entry point."""
    # Initialize logging ONLY when the CLI is actually invoked.
    configure_loggers()


@app.command(name="add")
def add_cmd(
    x: float = typer.Argument(..., help="First number"),
    y: float = typer.Argument(..., help="Second number"),
) -> None:
    """Add X and Y."""
    logger.info("CLI request: add %s + %s", x, y)
    result = add(x, y)
    logger.info("CLI success: %s + %s = %s", x, y, result)
    typer.echo(f"Result: {x} + {y} = {result}")


@app.command(name="subtract")
def subtract_cmd(
    x: float = typer.Argument(..., help="Number to subtract from"),
    y: float = typer.Argument(..., help="Number to subtract"),
) -> None:
    """Subtract Y from X."""
    logger.info("CLI request: subtract %s - %s", x, y)
    result = subtract(x, y)
    logger.info("CLI success: %s - %s = %s", x, y, result)
    typer.echo(f"Result: {x} - {y} = {result}")


@app.command(name="multiply")
def multiply_cmd(
    x: float = typer.Argument(..., help="First number"),
    y: float = typer.Argument(..., help="Second number"),
) -> None:
    """Multiply X by Y."""
    logger.info("CLI request: multiply %s * %s", x, y)
    result = multiply(x, y)
    logger.info("CLI success: %s * %s = %s", x, y, result)
    typer.echo(f"Result: {x} * {y} = {result}")


@app.command(name="divide")
def divide_cmd(
    x: float = typer.Argument(..., help="The numerator"),
    y: float = typer.Argument(..., help="The denominator"),
) -> None:
    """Divide X by Y."""
    logger.info("CLI request: divide %s / %s", x, y)
    try:
        result = div(x, y)
        logger.info("CLI success: %s / %s = %s", x, y, result)
        typer.echo(f"Result: {x} / {y} = {result}")
    except ZeroDivisionError as e:
        # Log as WARNING: it's a client input error, not a system crash
        logger.warning("CLI division error: %s", e)
        typer.echo("Error: Division by zero.", err=True)
        raise typer.Exit(code=1) from None


if __name__ == "__main__":
    app()

# src/tpp_mizan/core/logging.py
import logging
import logging.config
import os
from pathlib import Path

from dotenv import load_dotenv
from yaml import safe_load

_CONFIGURED = False  # <-- Guard against double initialization


def _default_project_root() -> Path:
    """Find the project root by walking up to pyproject.toml; fall back to cwd."""
    for parent in Path(__file__).resolve().parents:
        if (parent / "pyproject.toml").is_file():
            return parent
    return Path.cwd()


def configure_loggers() -> None:
    global _CONFIGURED
    if _CONFIGURED:
        return

    package_dir = Path(__file__).resolve().parent.parent
    config_path = package_dir / "config" / "my_log.yaml"

    with open(config_path) as f:
        config = safe_load(f)

    # Dynamically set log path to project root, OR override via env var
    project_root = _default_project_root()

    # Load environment variables from .env file in the project root
    load_dotenv(project_root / ".env", override=False)

    log_dir = Path(os.getenv("TPP_MIZAN_LOG_DIR", str(project_root / "logs")))
    log_dir.mkdir(parents=True, exist_ok=True)

    config["handlers"]["json_file"]["filename"] = str(log_dir / "tpp_mizan.json.log")
    config["handlers"]["timed_file"]["filename"] = str(log_dir / "tpp_mizan_timed.log")

    # Override log level from env
    env_log_level = os.getenv("LOG_LEVEL", "INFO").upper()
    config["loggers"]["tpp_mizan"]["level"] = env_log_level
    config["handlers"]["c_handler"]["level"] = env_log_level

    logging.config.dictConfig(config)
    _CONFIGURED = True

# Removed logger import and instantiation since we no longer need it here.


def add(x: float, y: float) -> float:
    """Add x and y."""
    return x + y


def subtract(x: float, y: float) -> float:
    """Subtract y from x."""
    return x - y


def multiply(x: float, y: float) -> float:
    """Multiply x by y."""
    return x * y


def div(x: float, y: float) -> float:
    """Divide x by y. Raises ZeroDivisionError if y is 0."""
    if y == 0:
        raise ZeroDivisionError("Cannot divide by zero")
    return x / y

import logging

import pytest

from tpp_mizan.core import logging as applog


@pytest.fixture(autouse=True)
def _isolate_logs(tmp_path, monkeypatch):
    # 1. Force logs to a unique temporary directory for this specific test
    monkeypatch.setenv("TPP_MIZAN_LOG_DIR", str(tmp_path / "logs"))

    # 2. Guard against a stray .env file overriding the log level during tests
    monkeypatch.delenv("LOG_LEVEL", raising=False)

    # 3. Allow configure_loggers() to run fresh
    applog._CONFIGURED = False

    yield

    # 4. Cleanup: Explicitly close and remove all handlers to prevent cross-test bleed
    pkg_logger = logging.getLogger("tpp_mizan")
    for handler in list(pkg_logger.handlers):
        handler.close()
        pkg_logger.removeHandler(handler)

    # 5. Reset the guard for the next test
    applog._CONFIGURED = False
