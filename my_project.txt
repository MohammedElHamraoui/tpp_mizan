# Define phony targets (targets that don't represent files)
.PHONY: help sync install test lint format run run-api build clean deep-clean context add add-dev

# Default target
help: ## Show this help message
	@echo "Usage: make [target]"
	@echo ""
	@echo "Targets:"
	@grep -E '^[a-zA-Z_-]+:.*?## .*$$' $(MAKEFILE_LIST) | awk 'BEGIN {FS = ":.*?## "}; {printf "  \033[36m%-15s\033[0m %s\n", $$1, $$2}'

# --- Dependency Management ---
sync: ## Create venv, resolve lockfile, and install project
	uv sync

install: sync ## Ensure project is installed (runs sync)
	@echo "Project synced and installed successfully."

add: ## Add a new dependency (Usage: make add PKG=requests)
	uv add $(PKG)

add-dev: ## Add a dev dependency (Usage: make add-dev PKG=httpx2)
	uv add --dev $(PKG)

remove: ## Remove a dependency (Usage: make remove PKG=requests)
	uv remove $(PKG)

lock: ## Update the lockfile without installing to the venv
	uv lock

# --- Code Quality ---
format: ## Format code using Ruff
	uv run ruff format .
	uv run ruff check --fix .

lint: ## Lint code using Ruff and Mypy
	uv run ruff check .
	uv run mypy .

# --- Testing ---
test: ## Run tests with pytest
	uv run pytest -v

test-cov: ## Run tests with coverage report
	uv run pytest --cov --cov-report=term-missing

# --- Execution ---
# Define default values that can be overridden from the command line
X ?= 10
Y ?= 2

run: ## Run the main CLI application (Usage: make run X=5 Y=2)
	uv run my-divide $(X) $(Y)

run-api: ## Run the FastAPI application with hot-reloading
	uv run uvicorn my_project.app:app --reload --host 127.0.0.1 --port 8000

# --- Maintenance ---
clean: ## Clean up caches and virtual environments
	rm -rf .venv .pytest_cache .mypy_cache .ruff_cache .coverage dist build
	find . -type d -name "__pycache__" -exec rm -rf {} +
	uv cache clean

deep-clean: clean ## Nuclear clean: removes logs, context file, and lockfile
	rm -rf logs/
	rm -f my_project.txt
	rm -f uv.lock
	@echo "Deep clean complete. (Next 'make install' will re-resolve dependencies)"

# --- CI/CD ---
ci-check: format lint test ## Run all CI checks locally

# --- LLM ---
context: ## Export project source to my_project.txt for LLM feedback
	@echo "Generating LLM context file..."
	@find . -type f \( -name "*.py" -o -name "*.toml" -o -name "*.md" -o -name "*.yaml" -o -name "Makefile" \) \
		-not -path "./.venv/*" -not -path "./logs/*" -not -path "./.git/*" \
		| sort | while read -r f; do \
			echo "========== $$f =========="; \
			cat "$$f"; \
			echo ""; \
		done > my_project.txt
	@echo "Done! Saved to my_project.txt ($$(wc -l < my_project.txt) lines, $$(du -h my_project.txt | cut -f1))"

# src/my_project/app.py
import logging

from fastapi import FastAPI, HTTPException

from my_project.core.logging import configure_loggers
from my_project.core.math_ops import div

# Initialize logging when the API module is loaded
logger = logging.getLogger(__name__)
configure_loggers()


app = FastAPI(title="My Project API")


@app.get("/")
def read_root():
    logger.info("Root endpoint accessed")
    return {"message": "Welcome to My Project API"}


@app.get("/divide/{x}/{y}")
def divide(x: float, y: float):
    logger.info("API request: divide %s by %s", x, y)
    try:
        result = div(x, y)
        return {"x": x, "y": y, "result": result}
    except ZeroDivisionError as e:
        logger.error("API error: %s", e)
        # Return a proper HTTP 400 error instead of a 500 server crash
        raise HTTPException(status_code=400, detail=str(e))

# src/my_project/cli.py
import logging

import typer

from my_project.core.logging import configure_loggers
from my_project.core.math_ops import div

# Initialize logging when the CLI starts
configure_loggers()
logger = logging.getLogger(__name__)

app = typer.Typer(help="A simple division CLI tool.")


@app.command()
def divide(
    x: float = typer.Argument(..., help="The numerator"),
    y: float = typer.Argument(..., help="The denominator"),
):
    """Divide X by Y."""
    logger.info("CLI request: divide %s by %s", x, y)  # <-- Add this
    try:
        result = div(x, y)
        logger.info("CLI success: %s / %s = %s", x, y, result)  # <-- Add this
        typer.echo(f"Result: {x} / {y} = {result}")
    except ZeroDivisionError as e:
        logger.error("CLI error: %s", e)  # <-- Add this
        typer.echo("Error: Division by zero.", err=True)
        raise typer.Exit(code=1)


if __name__ == "__main__":
    app()

# src/my_project/core/logging.py
import logging
import logging.config
import os
from pathlib import Path

from yaml import safe_load


def configure_loggers():
    package_dir = Path(__file__).resolve().parent.parent
    config_path = package_dir / "config" / "my_log.yaml"

    with open(config_path) as f:
        config = safe_load(f)

    # Dynamically set log path to project root
    project_root = package_dir.parent.parent
    log_dir = project_root / "logs"
    log_dir.mkdir(parents=True, exist_ok=True)
    config["handlers"]["json_file"]["filename"] = str(log_dir / "my_project.json.log")
    config["handlers"]["timed_file"]["filename"] = str(
        log_dir / "my_project_timed.log"
    )  # <-- Added this

    # Override log level from env
    env_log_level = os.getenv("LOG_LEVEL", "DEBUG").upper()
    config["loggers"]["my_project"]["level"] = env_log_level

    # Also override console handler level
    config["handlers"]["c_handler"]["level"] = env_log_level

    logging.config.dictConfig(config)

# src/my_project/core/math_ops.py
import logging

logger = logging.getLogger(__name__)


def div(x: float, y: float) -> float:
    """Divide x by y. Raises ZeroDivisionError if y is 0."""
    if y == 0:
        logger.error("A math error occurred: Division by zero!")
        raise ZeroDivisionError("Cannot divide by zero")
    return x / y
